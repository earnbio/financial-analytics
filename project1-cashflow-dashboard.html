# Project 1: Cash Flow Forecasting
# Commit: "Add data loading and cleaning script"

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from statsmodels.tsa.arima.model import ARIMA

# ── Step 1: Load Dataset ──────────────────
# Download from Kaggle: "Personal Finance Dataset"
df = pd.read_csv('transactions.csv')
df['date'] = pd.to_datetime(df['date'])
df = df.sort_values('date')

# ── Step 2: Clean & Categorise ───────────
categories = {
  'rent': 'Housing', 'mortgage': 'Housing',
  'food': 'Food', 'restaurant': 'Food',
  'uber': 'Transport', 'petrol': 'Transport'
}
df['category'] = df['description'].str.lower().map(
  lambda x: next((v for k, v in categories.items() 
    if k in x), 'Other'))

# ── Step 3: Monthly Aggregation ──────────
monthly = df.resample('ME', on='date').agg({
  'income': 'sum',
  'expense': 'sum'
})
monthly['net'] = monthly['income'] - monthly['expense']

# ── Step 4: ARIMA Forecast ───────────────
model = ARIMA(monthly['net'], order=(2, 1, 2))
result = model.fit()
forecast = result.forecast(steps=6)
print(forecast)
